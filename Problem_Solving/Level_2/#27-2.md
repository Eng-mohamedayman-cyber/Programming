//#27 — Calculate the Average of Array Elements
#include <iostream>
#include<string>
#include<cstdlib>
#include<ctime>
using namespace std;
int RandomNumber(int From , int To)
{
    int RanNum = rand() % (To - From + 1 ) + From ;
    return RanNum;
}
void FullArrayByRandomNumber(int Arr[100] ,int  &ArrLength)
{
    cout<<"\nEnter number of element : "<<endl;
    cin>>ArrLength;

    for(int i = 0 ; i < ArrLength ; i++ )
    {
        Arr[i]= RandomNumber(1,100);
    }
    cout<<endl;
}
void PrintArray(int Arr[100],int ArrLenggth )
{

    for(int i = 0 ; i < ArrLenggth ; i++)
    {
        cout<<Arr[i]<<" ";
    }
    cout<<endl;
}
int SumOfRandomArray(int Arr[100] , int ArrLength)
{
    int Sum = 0 ;
    for(int i = 0 ; i < ArrLength ; i++)
    {
        Sum += Arr[i] ;
    }
    return Sum ;
}
float AverageOfRandomArray(int Arr[100] , int ArrLength)
{
    return (float)SumOfRandomArray(Arr , ArrLength) / ArrLength ;
}
int main()
{
    srand((unsigned)time(NULL));

    int Arr[100] , ArrLength  ;

    FullArrayByRandomNumber(Arr,ArrLength);

    cout<<"Array Elements : ";
    PrintArray(Arr,ArrLength);

    cout<<"\nSum of all number is : ";
    cout<<SumOfRandomArray(Arr,ArrLength)<<endl;

    cout<<"\nAverage of all number is : ";
    cout<<AverageOfRandomArray(Arr,ArrLength)<<endl;

}
