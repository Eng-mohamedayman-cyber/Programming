//#4 — Print Perfect Numbers from 1 to N
#include <iostream>
#include<string>
#include<cmath>
using namespace std;
int ReadPositiveNumber(string Message)
{
    int Number ;
    do
    {
       cout<<Message<<endl;
       cin>>Number;
    }while(Number <= 0 );

    return Number ;
}
bool isPrefectNumber(int Number)
{
    int Sum = 0 ;
    for(int i = 1 ; i < Number ; i++ )
    {
        if(Number % i == 0)
        {
            Sum += i;
        }
    }
    return (Number == Sum) ;
}

void PrintResults(int Number )
{
    cout<<endl<<"The Prefect number from 1 to "<<Number<<" are : \n"<<endl;
    for(int i = 1 ; i <= Number ; i++)
    {
        if( isPrefectNumber(i))
        cout<<i<<endl;
    }

}
int main()
{
    PrintResults(ReadPositiveNumber("Please enter Positive Number ? "));
    return 0;
}
