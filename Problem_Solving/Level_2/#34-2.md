int ReadPositiveNumber(string Message)
{
    int Number ;
    do
    {
       cout<<Message<<endl;
       cin>>Number;
    }while(Number <= 0 );

    return Number ;
}
int RandomNumber(int From , int To)
{
    int RanNum = rand() % (To - From + 1 ) + From ;
    return RanNum;
}
void FullArrayByRandomNumber(int Arr[100] ,int  &ArrLength)
{
    cout<<"\nEnter number of element : "<<endl;
    cin>>ArrLength;

    for(int i = 0 ; i < ArrLength ; i++ )

}

{

    for(int i = 0 ; i < ArrLenggth ; i++)

    }

}
int FindNumberPositionInArray(int Number , int ArrLength , int Arr[100])
{
    for(int i = 0 ; i < ArrLength ; i++)
    {
        if( Arr[i] == Number )
        {
            return i ;
        }
    }
    return -1 ;
}
