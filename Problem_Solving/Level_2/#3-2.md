//#3 — Check if a Number is Perfect
#include <iostream>
#include<string>
#include<cmath>
using namespace std;
int ReadPositiveNumber(string Message)
{
    int Number ;
    do
    {
       cout<<Message<<endl;
       cin>>Number;
    }while(Number <= 0 );

    return Number ;
}
bool isPrefectNumber(int Number)
{
    int Sum = 0 ;
    for(int i = 1 ; i < Number ; i++ )
    {
        if(Number % i == 0)
        {
            Sum += i;
        }
    }
    return (Number == Sum) ;
}

void PrintResults(int Number )
{
    if( isPrefectNumber(Number))
        cout<<Number<<" is Prefect Number "<<endl;
    else
        cout<<Number<<" is Not Prefect Number "<<endl;
}
int main()
{
    PrintResults(ReadPositiveNumber("Please enter Positive Number ? "));
    return 0;
}
