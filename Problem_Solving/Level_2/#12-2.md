//#12 — Print an Inverted Number Pattern
#include <iostream>
#include<string>
using namespace std;
int ReadPositiveNumber(string Message)
{
    int Number ;
    do
    {
       cout<<Message<<endl;
       cin>>Number;
    }while(Number <= 0 );

    return Number ;
}
void PrintInvertedNumberPattern(int Number)
{
    cout<<endl;
    for(int i = Number ; i > 0 ; i-- )
    {
        for(int j = 1 ; j <=  i ; j++ )
        {
            cout<<i;
        }
        cout<<endl;
    }
}
int main()
{
    PrintInvertedNumberPattern(ReadPositiveNumber("Please enter positive number? "));
    return 0;
}
