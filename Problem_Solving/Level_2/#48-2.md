//#48 — Implement the Floor Function
#include <iostream>
#include<cmath>
using namespace std;
float MyFloor(float Number)
{
    int IntPart;
    IntPart = int(Number);

    if(Number > 0 )
    {
        return IntPart ;
    }
    else
    {
        return --IntPart;
    }
}
float ReadNumber()
{
    float Number ;

    cout<<"Please enter the number? ";
    cin>>Number;

    return Number ;
}
int main()
{
    float Number = ReadNumber() ;

    cout<<"My Floor Result : "<<MyFloor(Number)<<endl;
    cout<<"C++ floor Result : "<<floor(Number)<<endl;

    return 0;
}
