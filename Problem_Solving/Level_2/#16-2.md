```c++
//#16 — Generate All Three-Letter Combinations from AAA to ZZZ
#include <iostream>
#include<string>
using namespace std;
void PrintFrom_AAA_To__ZZZ()
{
    cout<<endl;
    string word = "";
    for(int i = 65 ; i <= 90 ; i++ )
    {
        for(int j = 65 ; j <= 90 ; j++ )
        {
            for(int k = 65 ; k <= 90 ; k++)
            {
                word = word + char(i);
                word = word + char(j);
                word = word + char(k);
                
                cout<<word<<endl;

                word = "";
            }
        }
        cout<<"_________________________________________________"<<endl;
    }
}
int main()
{
    PrintFrom_AAA_To__ZZZ();
    return 0;
}
```
