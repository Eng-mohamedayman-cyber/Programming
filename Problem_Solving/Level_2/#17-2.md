//#17 — Brute-Force Password Guessing
#include <iostream>
#include<string>
using namespace std;
string ReadLetterPassword()
{
    string Password;

    cout<<"Enter Password From Three Capital Letter? "<<endl;
    cin>>Password;

    return Password;
}
bool GuessPassword(string Password)
{
    string word = "";
    int Counter = 0;

    cout<<endl;
    for(int i = 65 ; i <= 90 ; i++ )
    {
        for(int j = 65 ; j <= 90 ; j++ )
        {
            for(int k = 65 ; k <= 90 ; k++)
            {
                Counter++;

                word = word + char(i);
                word = word + char(j);
                word = word + char(k);

                cout<<"Trial ["<<Counter<<"] :"<<word<<endl;

                if(word == Password)
                {
                    cout<<"\nPassword is "<<word<<"\n";
                    cout<<"Found after "<<Counter<<endl;
                    return true ;
                }
                word = "";
            }
        }
    }
    return false;
}
int main()
{
    GuessPassword(ReadLetterPassword());
    return 0;
}
