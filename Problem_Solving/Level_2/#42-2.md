//#42 — Count Odd Numbers in an Array
#include<iostream>
#include<cstdlib>
#include<ctime>
#include<string>
#include<cmath>
using namespace std;
int RandomNumber(int From , int To)
{
    int RanNum = rand() % (To - From + 1 ) + From ;
    return RanNum;
}
void FullArrayByRandomNumber(int Arr[100] ,int  &ArrLength)
{
    cout<<"\nEnter number of element : "<<endl;
    cin>>ArrLength;

    for(int i = 0 ; i < ArrLength ; i++ )
    {
        Arr[i]= RandomNumber(1,100);
    }
    cout<<endl;
}
void PrintArray(int Arr[100],int ArrLenggth )
{

    for(int i = 0 ; i < ArrLenggth ; i++)
    {
        cout<<Arr[i]<<" ";
    }
    cout<<endl;
}
int OddCounter(int Arr[100],int ArrLength)
{
    int Counter = 0 ;

    for(int i = 0 ; i < ArrLength ; i++ )
    {

        if(Arr[i] % 2 == 1)
        {
            Counter++;
        }
    }

    return Counter ;
}
int main()

    cout<<"\nArray 1 Element: \n" ;
    PrintArray(Arr,ArrLength);
    cout<<"\nOdd Number Counter = ";
    cout<<OddCounter(Arr,ArrLength)<<endl;

}
