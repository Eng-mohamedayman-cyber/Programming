//#19 — Generate Random Numbers Within a Range
#include <iostream>
#include<string>
#include<ctime>
#include<cstdlib>
using namespace std;
int RandomNumber(int From , int To)
{
    int RanNum = rand() % (To - From + 1 ) + From ;
    return RanNum;
}
int main()
{
    srand((unsigned)time(NULL));

    cout<<RandomNumber(1,20)<<endl;
    cout<<RandomNumber(1,20)<<endl;
    cout<<RandomNumber(1,20)<<endl;
    return 0;
}
