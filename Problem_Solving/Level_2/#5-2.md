//#5 — Print the Digits of a Number
#include<iostream>
#include<string>
using namespace std;
int ReadPositiveNumber(string Message)
{
    int Number ;
    do
    {
        cout<<Message<<endl;
        cin>>Number;
    }while(Number <= 0);
    return Number;
}
void PrintDigits(int Number)
{
    int Remainder ;
    while(Number > 0 )
    {
        Remainder = Number % 10 ;
        Number /= 10 ;
        cout<<Remainder<<endl;
    }
}
int main()
{
    PrintDigits(ReadPositiveNumber("Please enter Positive number ? "));
    return 0 ;
}
