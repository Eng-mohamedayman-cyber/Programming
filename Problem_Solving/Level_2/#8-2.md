//#8 — Count Digit Frequency in a Number
#include <iostream>
#include<string>
using namespace std;
int ReadPositiveNumber(string Message)
{
    int Number ;
    do
    {
       cout<<Message<<endl;
       cin>>Number;
    }while(Number <= 0 );

    return Number ;
}
int CountDigitsFrequency( short DigitToCheck ,int Number )
{
    int Remainder = 0 , FreqCount = 0 ;
    while( Number > 0 )
    {
        Remainder = Number % 10 ;
        Number /= 10 ;
        if(Remainder == DigitToCheck)
        {
            FreqCount++;
        }
    }
    return FreqCount;
}
int main()
{
    int Number = ReadPositiveNumber("Please enter main number ? ");
    int DigitToCheck = ReadPositiveNumber("Please Enter one Digit To Check ? ") ;
    cout<<"\nDigit "<<DigitToCheck<<" Frequency is "<<CountDigitsFrequency(DigitToCheck,Number)<<" Time(s) "<<endl;
    return 0;
}
