#include <iostream>

using namespace std;
enum enPrimeNotPrime{ Prime = 1 , NotPrime = 2 };
enPrimeNotPrime chackPrime(int Number)
{
    int M =  round(Number / 2);
    for(int Counter = 2 ; Counter <= M ; Counter++ )
    {
       if( Number % Counter == 0 )
            return enPrimeNotPrime::NotPrime;
    }

    return enPrimeNotPrime::Prime;
}
int RandomNumber(int From , int To)
{
    int RanNum = rand() % (To - From + 1 ) + From ;
    return RanNum;
}

{
    cout<<"\nEnter number of element : "<<endl;
    cin>>ArrLength;

    for(int i = 0 ; i < ArrLength ; i++ )
    {

}

void CopyPrimeNumberQnlyInArray(int ArrSource[100] , int ArrDestination[100] , int ArrLength , int& Arr2Length )
{
    Arr2Length = 0 ;
    for(int i = 0 ; i < ArrLength ; i++)
    {
        if(chackPrime(ArrSource[i]) == enPrimeNotPrime::Prime)
        {
            ArrDestination[Arr2Length] = ArrSource[i];
            Arr2Length++;
        }
    }
}
