//#2 — Read and Return a String Using a Function
#include <iostream>
#include<string>
using namespace std;
string ReadName()
{
    string name;
    cout<<"Please enter your name "<<endl;
    getline(cin,name);
    return name;
}
void PrintName(string name)
{
    cout<<"\n your name is : "<<name<<endl;
}
int main()
{
    PrintName(ReadName());
    return 0;
}
