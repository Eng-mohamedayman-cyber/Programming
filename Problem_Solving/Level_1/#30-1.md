//#30 — Calculate the Factorial of a Number
#include<iostream>
#include<string>
using namespace std;
int ReadPositiveNumber(string Message)
{
  int N ;
  do
  {
      cout<<Message<<endl;
      cin>>N;
  }while(N<0);
  return N ;
}
int Factorial(int N)
{
    int fac = 1 ;
    for(int i = N ; i>=1 ; i-- )
    {
        fac = fac * i ;
    }
    return fac ;
}
int main()
{
    cout<<Factorial(ReadPositiveNumber("Enter N ?"))<<endl;
    return 0 ;
}
