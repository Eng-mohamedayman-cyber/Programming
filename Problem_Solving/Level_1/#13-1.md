{

    cout<<"Please enter the number 3 : ";
    cin>>num3;

{
    if(num1 > num2 )
    {
        if(num1>num3)
        {
            return num1;
        }
        else
        {
            return num3;
        }
    }
    else
    {
        if(num2>num3)
        {
            return num2;
        }
        else
        {
            return num3;
        }
    }
}
void PrintResult(int Max)
{
    cout<<endl<<"The Maximum Number is : "<<Max<<endl;
}
int main()
{
    int num1 , num2 , num3;
