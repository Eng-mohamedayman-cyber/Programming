#include<iostream>
#include<string>
using namespace std;
struct stInfo
{
   int age;
   bool HasDriverlicense;
   bool HasRecommendation;
};
stInfo ReadInfo()
{
    stInfo Info;
    cout<<"please  enter  your age : ";
    cin>>Info.age;
    cout<<"having driver license : ";
    cin>>Info.HasDriverlicense;
    cout<<"having Recommendation : ";
    cin>>Info.HasRecommendation;
    return Info;
}
bool IsAccepted(stInfo Info)
{
    if(Info.HasRecommendation)
    {
        return true;
    }
    else
    {
        return (Info.age>21 && Info.HasDriverlicense);
    }
}
void PrintInfo(stInfo Info)
{
    if(IsAccepted(Info))
    {
        cout<<"\nHired"<<endl;
    }
    else
    {
        cout<<"\nRejected"<<endl;
    }
}
int main()
{
    PrintInfo(ReadInfo());
    return 0;
}
