#23 — Calculate the Area of the Circumcircle of a Triangle
#include<iostream>
#include<string>
#include<cmath>
using namespace std;
void ReadTrianleData(float &A , float &B , float &C)
{
    cout<<"Please enter the A? "<<endl;
    cin>>A;
    cout<<"Please enter the B? "<<endl;
    cin>>B;
    cout<<"Please enter the C? "<<endl;

}
float CircleAreaByTriangle(float A ,float B , float C)
{
    float Area , P , T;
    const float PI = 3.141592 ;
    P = (A+B+C)/2;
    T = (A*B*C)/(4*sqrt(P*(P-A)*(P-B)*(P-C)));
    Area = PI * pow(T,2);
    return Area ;
}
void PrintResult(float Area)
{
    cout<<endl<<"The area = "<<Area<<endl;
}
int main()
{
    float A,B,C;
    ReadTrianleData(A,B,C);
    PrintResult(CircleAreaByTriangle(A,B,C));
    return 0 ;
}
