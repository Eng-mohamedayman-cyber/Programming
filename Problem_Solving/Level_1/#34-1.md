//#34 — Calculate Sales Commission Based on Total Sales
#include<iostream>
#include<string>
using namespace std;
int ReadTotalSales()
{
    int TotalSales ;
    cout<<"Please enter Total Sales ? "<<endl;
    cin>>TotalSales;
    return TotalSales;
}
float GetComissionPercentsge(float TotalSales)
{
    if(TotalSales >= 1000000)
        return 0.01;
    else if(TotalSales >= 500000)
        return 0.02;
    else if(TotalSales >= 100000)
        return 0.03;
    else if(TotalSales >= 50000)
        return 0.05;
    else
        return 0.00;
}
float calculateTotalComission(float TotalSales)
{
    return GetComissionPercentsge(TotalSales)*TotalSales;
}
int main()
{
    float TotalSales = ReadTotalSales();
    cout<<endl<<"Comission Percentage = "<<GetComissionPercentsge(TotalSales)<<endl;
    cout<<endl<<"Total Comission = "<<calculateTotalComission(TotalSales)<<endl;
    return 0 ;
}
