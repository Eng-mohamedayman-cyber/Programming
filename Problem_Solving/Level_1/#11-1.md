//#11 — Calculate Average and Check Pass or Fail
#include<iostream>
#include<string>
using namespace std;
enum enPassFail{Pass=1 , Fail = 2};
int ReadMark(int &Mark1 , int &Mark2 , int &Mark3)
{
    cout<<"Please enter the Mark 1 : ";
    cin>>Mark1;
    cout<<"Please enter the Mark 2 : ";
    cin>>Mark2;
    cout<<"Please enter the Mark 3 : ";
    cin>>Mark3;
}
int SumOf3Number(int Mark1 , int Mark2 , int Mark3 )
{
    return Mark1 + Mark2 + Mark3 ;
}
float CalculateAverage(int Mark1 , int Mark2 , int Mark3 )
{
    return (float)SumOf3Number( Mark1 , Mark2 , Mark3 )/3 ;
}
enPassFail ChackAverage(int Average)
{
    if(Average>=50)
    {
        return enPassFail::Pass;
    }
    else
    {
        return enPassFail::Fail;
    }
}
void PrintResult(float Average)
{
    cout<<endl<<"The average of the Mark : "<<Average<<endl;
    if(ChackAverage(Average) == enPassFail::Pass)
    {
        cout<<"you are Pass"<<endl;
    }
    else
    {
        cout<<"you are Fail"<<endl;
    }
}
int main()
{
    int Mark1 , Mark2 , Mark3 ;
    ReadMark( Mark1 , Mark2 , Mark3 );
    PrintResult(CalculateAverage( Mark1 , Mark2 , Mark3 ));

    return 0;
}
