//#8 — Check Student Pass or Fail
#include<iostream>
#include<string>
using namespace std;
enum enPassFail{Pass=1 , Fail = 2};
int ReadMark()
{
    int Mark;
    cout<<"Please enter your mark : ";
    cin>>Mark;
    return Mark;
}
enPassFail ChackMark(int Mark)
{
    if(Mark>=50)
    {
        return enPassFail::Pass;
    }
    else
    {
        return enPassFail::Fail;
    }
}
void PrintChackMark(int Mark)
{
    if(ChackMark(Mark) == enPassFail::Pass)
    {
        cout<<endl<<"you are Pass"<<endl;
    }
    else
    {
        cout<<endl<<"you are Fail"<<endl;
    }
}
int main()
{
    PrintChackMark(ReadMark());
    return 0;
}
