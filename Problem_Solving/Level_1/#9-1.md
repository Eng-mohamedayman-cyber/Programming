//#9 — Calculate the Sum of Three Numbers
#include<iostream>
#include<string>
using namespace std;
void ReadNumber(int &num1 , int &num2 , int &num3)
{
    cout<<"Please enter the number 1 : ";
    cin>>num1;
    cout<<"Please enter the number 2 : ";
    cin>>num2;
    cout<<"Please enter the number 3 : ";
    cin>>num3;
}
int SumOf3Number(int num1 , int num2 , int num3 )
{
    return num1 + num2 + num3 ;
}
void PrintResult(int Total)
{
    cout<<"The total sum of the number : "<<Total<<endl;
}
int main()
{
    int num1 , num2 , num3 ;
    ReadNumber(num1,num2,num3);
    PrintResult(SumOf3Number(num1,num2,num3));
}
