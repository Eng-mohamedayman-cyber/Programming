//#41 — Convert Hours to Days and Weeks
#include<iostream>
#include<string>
#include<cmath>
using namespace std;
float ReadPositiveNumber(string Message)
{
    int Number ;
    do
    {
       cout<<Message<<endl;
       cin>>Number;
    }while(Number <= 0 );

    return Number ;
}
float HoursToDays(float NumberOfHours)
{
    return (float)NumberOfHours / 24 ;
}
float HoursToWeek(float NumberOfHours)
{
    return (float)NumberOfHours / 24 / 7 ;
}
float DaysToWeek(float NumberOfDays)
{
    return (float)NumberOfDays / 7 ;
}
int main()
{
    float NumberOfHours = ReadPositiveNumber("Please enter number of Hours? ");
    float NumberOfDays = HoursToDays(NumberOfHours);
    float NumberOfWeek = DaysToWeek(NumberOfDays);

    cout<<endl<<"The Hours = "<<NumberOfHours<<endl;
    cout<<"The Days = "<<NumberOfDays<<endl;
    cout<<"The Week = "<<NumberOfWeek<<endl;
    return 0 ;
}
