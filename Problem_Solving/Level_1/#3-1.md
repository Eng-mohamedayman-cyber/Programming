//#3 — Check if a Number is Odd or Even
#include<iostream> 
#include<string> 
using namespace std; 
enum enNumberType{odd = 1 , even = 2}; 
int readnumber() 
{ 
    int num; 
    cout<<"Please enter the number "<<endl; 
    cin>>num; 
    return num; 
} 
enNumberType ChackNumberType(int num) 
{ 
    int result = num % 2 ; 
    if(result == 0) 
    { 
       return enNumberType::even; 
    } 
    else 
    { 
        return enNumberType::odd; 
    } 
} 
void PrintOddorEven(enNumberType numbertype) 
{ 
    if(numbertype == enNumberType::even) 
    { 
       cout<<"The number is even "<<endl; 
    } 
    else 
    { 
       cout<<"The number is odd "<<endl; 
    } 
} 
int main() 
{ 
    PrintOddorEven(ChackNumberType(readnumber())); 
    return 0; 
}
