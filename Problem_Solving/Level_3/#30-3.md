//#30-3 - Count the Number of Occurrences of a Specific Character in a String
#include <iostream>
#include <cmath>
#include <string>

using namespace std;

string ReadString()
{
    string S1;
    cout << "\nPlease Enter Your String? " << endl;
    getline(cin, S1);

    return S1;
}
char ReadChar()
{
    char C1;
    cout << "\nPlease Enter Your Char? " << endl;
    cin >> C1;

    return C1;
}

short CountLetter(string S1, char C1)
{
    short counter = 0;

    for (short i = 0; i <= S1.length(); i++)
    {
        if (S1[i] == C1)
            counter++;
    }

    return counter;
}
int main()
{
    string S1 = ReadString();
    char C1 = ReadChar();

    cout << "\nLetter \'" << C1 << "\' Count : " << CountLetter(S1, C1) << endl;

    return 0;
}
