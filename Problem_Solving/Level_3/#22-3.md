// 22 - Print Fibonacci Series Using Recursion
#include <iostream>
#include <cmath>

using namespace std;

void PrintFibonacciUsingRecursion(short Number, short Prev1, short Prev2)
{
    int FibNumber = 0;
    
    if (Number > 0)
    {
        FibNumber = Prev1 + Prev2;
        Prev2 = Prev1;
        Prev1 = FibNumber;
        cout << FibNumber << "\t";

        return PrintFibonacciUsingRecursion(Number - 1, Prev1, Prev2);
    }
}
int main()
{
    PrintFibonacciUsingRecursion(10,0,1);

    return 0;
}
