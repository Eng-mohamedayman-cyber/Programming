// 18 - Print Intersected Numbers in Two Matrices
#include <iostream>
#include <cmath>
#include <cstdio>

using namespace std;

void PrintMatrix(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            cout<<arr[i][j]<<"\t";
        }
        cout << endl;
    }
}
bool IsNumberInMatrix(int arr[3][3], int Rows, int Cols, int Num)
{

    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Cols; j++)
        {
            if (arr[i][j] == Num)
            {
                return true;
            }
        }
    }

    return false;
}
void PrinrtIntersectedNumber(int Matrix1[3][3], int Matrix2[3][3], int Rows, int Cols)
{
    int Number;

    cout << "\nIntersected Numbers are : \n";

    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Cols; j++)
        {
            Number = Matrix1[i][j];

            if (IsNumberInMatrix(Matrix2, 3, 3, Number))
            {
                cout << Number << "\t";
            }
        }
    }
}
int main()
{
    int Matrix1[3][3] = { {77,5,12},{22,20,1},{1,0,9} }, Matrix2[3][3] = { {5,80,90},{22,77,1},{10,8,33} };

    cout << "Matrix 1 : " << endl;
    PrintMatrix(Matrix1, 3, 3);

    cout << "Matrix 1 : " << endl;
    PrintMatrix(Matrix2, 3, 3);

    PrinrtIntersectedNumber(Matrix1, Matrix2, 3, 3);

    return 0;
}
