// 26 - Convert All String Characters to Uppercase and Lowercase
#include <iostream>
#include <cmath>
#include <string>

using namespace std;

char ReadChar()
{
    char C1;
    cout << "Please Enter Your Char? " << endl;
    cin >> C1;

    return C1;
}

char InvertCharCase(char C1)
{
    return isupper(C1) ? tolower(C1) : toupper(C1);
}

int main()
{
    char C1 = ReadChar();
    cout << "\nChar After inverting case : " << endl;

    C1 = InvertCharCase(C1);
    cout << C1 << endl;

    return 0;
}
