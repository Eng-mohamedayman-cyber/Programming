```c++
// 21 - Print Fibonacci Series Using Loop
#include <iostream>
#include <cmath>

using namespace std;

void PrintFibonacciUsingLoop(short Number)
{
    int FibNumber = 0;
    int Prev2 = 0, Prev1 = 1;

    cout << "1\t";
    for (short i = 2; i <= Number; i++)
    {
        FibNumber = Prev1 + Prev2;
        cout << FibNumber << "\t";
        Prev2 = Prev1;
        Prev1 = FibNumber;
    }
}
int main()
{
    PrintFibonacciUsingLoop(10);

    return 0;
}
```
