// 12 - Check If Two Matrices Are Identical
#include <iostream>
#include <cmath>
#include <cstdio>

using namespace std;

int RandomNumber(int From, int To)
{
    int RanNum = rand() % (To - From + 1) + From;
    return RanNum;
}
void FillMatrixWithRandomNumber(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            arr[i][j] = RandomNumber(1, 10);
        }
    }
}
void PrintMatrix(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            printf(" %0*d", 2, arr[i][j]);
        }
        cout << endl;
    }
}
bool AreMatrixTypical(int Matrix1[3][3], int Matrix2[3][3], int Rows, int Cols)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Cols; j++)
        {
            if (Matrix1[i][j] != Matrix2[i][j])
            {
                return false;
            }
        }
    }
    return true;
}

int main()
{
    srand((unsigned)time(NULL));

    int Matrix1[3][3], Matrix2[3][3];

    FillMatrixWithRandomNumber(Matrix1, 3, 3);
    cout << "\nMatrix 1 : " << endl;
    PrintMatrix(Matrix1, 3, 3);

    FillMatrixWithRandomNumber(Matrix2, 3, 3);
    cout << "\nMatrix 2 : " << endl;
    PrintMatrix(Matrix2, 3, 3);

    if (AreMatrixTypical(Matrix1, Matrix2, 3, 3))
    {
        cout << "Yes : Matrix Is Typical " << endl;
    }
    else
    {
        cout << "No : Matrix Is NOT Typical " << endl;
    }

    return 0;
}
