//#44-3 - Remove All Punctuation Characters from a String While Preserving Letters, Numbers, and Spaces
#include <iostream>
#include <cmath>
#include <string>
#include <vector>

using namespace std;

string RemovePunctuationsFromString(string S1)
{
	string S2 = "";

	for (short i = 0; i < S1.length(); i++)
	{
		if (!ispunct(S1[i]))
		{
			S2 += S1[i];
		}
	}

	return S2;
}
int main()
{
	string S1 = "Welcome to Jordan, Jordan is a nice country; ...it's amazing.";
	
	cout << "\nOrigial string : \n" << S1;
	cout << "\nPunctuation Remove : \n" << RemovePunctuationsFromString(S1);
	return 0;
}
