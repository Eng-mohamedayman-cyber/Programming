// 16 - Check If a Matrix Is Sparse
#include <iostream>
#include <cmath>
#include <cstdio>

using namespace std;

int RandomNumber(int From, int To)
{
    int RanNum = rand() % (To - From + 1) + From;
    return RanNum;
}
void FillMatrixWithRandomNumber(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            arr[i][j] = RandomNumber(0, 3);
        }
    }
}
void PrintMatrix(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            printf("%0*d ", 2, arr[i][j]);
        }
        cout << endl;
    }
}
int CountOfNumberInMatrix(int arr[3][3], int Rows, int Cols, int Num)
{
    int Count = 0;

    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Cols; j++)
        {
            if (arr[i][j] == Num)
            {
                Count++;
            }
        }
    }

    return Count;
}
bool IsMatrixSparse(int arr[3][3], int Rows, int Cols)
{
    short MatrixSize = Rows * Cols;

    return (CountOfNumberInMatrix(arr,3,3,0)>(MatrixSize/2));
}
int main()
{
    srand((unsigned)time(NULL));

    int Matrix1[3][3];

    FillMatrixWithRandomNumber(Matrix1, 3, 3);
    cout << "Matrix 1 : " << endl;
    PrintMatrix(Matrix1, 3, 3);

    if (IsMatrixSparse(Matrix1, 3, 3))
    {
        cout << "Yes: Matrix is sparse\n";
    }
    else
    {
        cout << "No: Matrix is not sparse\n";
    }
    
    return 0;
}
