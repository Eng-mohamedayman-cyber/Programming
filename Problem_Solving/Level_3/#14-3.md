// 14 - Check If a Matrix Is Scalar
#include <iostream>
#include <cmath>
#include <cstdio>

using namespace std;

void PrintMatrix(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            printf("%0*d ", 2, arr[i][j]);
        }
        cout << endl;
    }
}
bool AreMatrixScalar(int Matrix1[3][3], int Rows, int Cols)
{
    int FirstDiagElemement = Matrix1[0][0];
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Cols; j++)
        {
            if ((i == j) && (Matrix1[i][j] != FirstDiagElemement))
            {
                return false;
            }
            else if ((i != j) && (Matrix1[i][j] != 0))
            {
                return false;
            }
        }
    }
    return true;
}

int main()
{
    int Matrix1[3][3] = { {9,0,0},{0,9,0},{0,0,9} };

    cout << "Matrix 1 : " << endl;
    PrintMatrix(Matrix1, 3, 3);


    if (AreMatrixScalar(Matrix1, 3, 3))
    {
        cout << "\nYes : Matrix Is Scalar " << endl;
    }
    else
    {
        cout << "\nNo : Matrix Is NOT Scalar " << endl;
    }

    return 0;
}
