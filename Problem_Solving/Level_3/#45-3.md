//#45-3 - Convert a Client Record Structure into a Delimited String Line for Data Storage and Processing
#include <iostream>
#include <string>
#include <vector>
#include <cmath>

using namespace std;

struct stInfoClient
{
	string AccountNumber;
	string PinCode;
	string Name;
	string NumberPhone;
	double AccountBalance;
};

stInfoClient ReadNewClient()
{
	stInfoClient Client;

	cout << "Enter Account Number? ";
	getline(cin,Client.AccountNumber);
	
	cout << "Enter Pin Code? ";
	getline(cin, Client.PinCode);

	cout << "Enter Name? ";
	getline(cin, Client.Name);

	cout << "Enter Number Phone? ";
	getline(cin, Client.NumberPhone);

	cout << "Enter Account Balance? ";
	cin >> Client.AccountBalance;

	return Client;
}
string InvertRecordToLine(stInfoClient Client, string Seperator = "#//#")
{
	string stClientRecord = "";

	stClientRecord += Client.AccountNumber + Seperator;
	stClientRecord += Client.PinCode + Seperator;
	stClientRecord += Client.Name + Seperator;
	stClientRecord += Client.NumberPhone + Seperator;
	stClientRecord += to_string(Client.AccountBalance);

	return stClientRecord;
}
int main()
{
	cout << "Enter Data Of Client : \n" << endl;

	stInfoClient Client1 = ReadNewClient();
	
	cout << "\n\n" << InvertRecordToLine(Client1) << "\n";
	
	return 0;
}
