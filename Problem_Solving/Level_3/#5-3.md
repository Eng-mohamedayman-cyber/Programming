// 05 - Sum of Each Column in a Matrix
#include <iostream>
#include <cmath>

using namespace std;

int RandomNumber(int From, int To)
{
    int RanNum = rand() % (To - From + 1) + From;
    return RanNum;
}
void FillMatrixWithRandomNumber(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            arr[i][j] = RandomNumber(1, 100);
        }
    }
}
void PrintMatrix(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            cout << arr[i][j] << "\t";
        }
        cout << endl;
    }
}
int ColsSum(int arr[3][3], int Row, int ColsNumber)
{
    int sum = 0;

    for (int i = 0; i < Row; i++)
    {
        sum = sum + arr[i][ColsNumber];
    }
    return sum;
}
void SumOfColsInArray(int arr[3][3],int ColSum[3],int Rows ,int Cols)
{
    for (int i = 0; i < Rows; i++)
    {
        ColSum[i] = ColsSum(arr, Rows, i);
    }
}
void PrintEachColsSum(int arr[3][3], int ColSum[3], int Rows, int Cols)
{
    cout << "The following are the sum of each Cols in the matrix : " << endl;

    for (int j = 0; j < Rows; j++)
    {
        cout << "Cols " << j + 1 << " sum : " << ColSum[j] << endl;
    }
}
int main()
{
    srand((unsigned)time(NULL));

    int arr[3][3];
    int ColSum[3];

    FillMatrixWithRandomNumber(arr, 3, 3);

    cout << "The following is a 3*3 random matrix : " << endl;
    PrintMatrix(arr, 3, 3);

    SumOfColsInArray(arr, ColSum, 3, 3);

    PrintEachColsSum(arr,ColSum, 3, 3);

    return 0;
}
