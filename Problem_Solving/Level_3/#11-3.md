// 11 - Check If Two Matrices Are Equal
#include <iostream>
#include <cmath>
#include <cstdio>

using namespace std;

int RandomNumber(int From, int To)
{
    int RanNum = rand() % (To - From + 1) + From;
    return RanNum;
}
void FillMatrixWithRandomNumber(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            arr[i][j] = RandomNumber(1, 10);
        }
    }
}
void PrintMatrix(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            printf(" %0*d", 2, arr[i][j]);
        }
        cout << endl;
    }
}
int  SumOfAllMatrix(int arr[3][3], int Rows, int Cols)
{
    int sum = 0;

    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Cols; j++)
        {
            sum = sum + arr[i][j];
        }
    }

    return sum;
}
bool AreMatrixEqual(int Matrix1[3][3], int Matrix2[3][3], int Rows, int Cols)
{
    return (SumOfAllMatrix(Matrix1, Rows, Cols) == SumOfAllMatrix(Matrix2, Rows, Cols));
}

int main()
{
    srand((unsigned)time(NULL));

    int Matrix1[3][3],Matrix2[3][3];

    FillMatrixWithRandomNumber(Matrix1, 3, 3);
    cout << "\nMatrix 1 : " << endl;
    PrintMatrix(Matrix1, 3, 3);

    FillMatrixWithRandomNumber(Matrix2, 3, 3);
    cout << "\nMatrix 2 : " << endl;
    PrintMatrix(Matrix2, 3, 3);

    if (AreMatrixEqual(Matrix1, Matrix2, 3, 3))
    {
        cout << "Yes : Matrix Is Equal " << endl;
    }
    else
    {
        cout << "No : Matrix Is NOT Equal " << endl;
    }

    return 0;
}
