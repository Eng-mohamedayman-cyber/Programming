//#36-3 - Count the Number of Words in a String
#include <iostream>
#include <cmath>
#include <string>

using namespace std;

string ReadString()
{
    string S1;
    cout << "Please Enter Your String? " << endl;
    getline(cin, S1);

    return S1;
}
short CountWord(string S1)
{
    string delim = " ";

    short pos = 0;
    string sWord;

    short count = 0;

    while ((pos = S1.find(delim)) != std::string::npos)
    {
        sWord = S1.substr(0, pos);
        if (sWord != "")
        {
            count++;
        }
        S1.erase(0, pos + delim.length());
    }

    if (S1 != "")
    {
        count++;
    }

    return count;

}

int main()
{
    string S1 = ReadString();

    cout<< "\nThe Number of word in string : \n" << CountWord(S1);

    return 0;
}
