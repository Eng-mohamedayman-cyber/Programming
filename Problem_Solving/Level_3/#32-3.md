```c++
//#32-3 - Check Whether a Character Is a Vowel or Not
#include <iostream>
#include <cmath>
#include <string>

using namespace std;

char ReadChar()
{
    char C1;
    cout << "\nPlease Enter Your Char? " << endl;
    cin >> C1;

    return C1;
}
bool IsVowel(char Letter)
{
    Letter = tolower(Letter);

    return ((Letter == 'a') || (Letter == 'e') || (Letter == 'u') || (Letter == 'i') || (Letter == 'o'));
}
int main()
{
    char C1 = ReadChar();

    if (IsVowel(C1))
        cout << "YES Letter \'" << C1 << "\' is vowel" << endl;
    else
        cout << "No Letter \'" << C1 << "\' is not vowel" << endl;

    return 0;
}
```
