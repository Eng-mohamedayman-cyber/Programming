// 07 - Transpose a Matrix
#include <iostream>
#include <cmath>

using namespace std;

void FillMatrix(int arr[3][3], int Rows, int Clos)
{
    int Counter = 1;

    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            arr[i][j] = Counter;
            Counter++;
        }
    }
}
void TransportMatrix(int arr[3][3],int arrTransport[3][3], int Rows, int Clos)
{
    int Counter = 1;

    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            arrTransport[i][j] = arr[j][i];
        }
    }
}
void PrintMatrix(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            cout << arr[i][j] << "\t";
        }
        cout << endl;
    }
}

int main()
{
    int arr[3][3],arrTransport[3][3];

    FillMatrix(arr, 3, 3);

    cout << "The following is a 3*3 order matrix : " << endl;
    PrintMatrix(arr, 3, 3);

    TransportMatrix(arr,arrTransport, 3, 3);

    cout << "The following is a 3*3 transport matrix : " << endl;
    PrintMatrix(arrTransport, 3, 3);

    return 0;
}
