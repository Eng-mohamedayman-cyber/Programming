// 19 - Find Maximum and Minimum Numbers in a Matrix
#include <iostream>
#include <cmath>
#include <cstdio>

using namespace std;

int RandomNumber(int From, int To)
{
    int RanNum = rand() % (To - From + 1) + From;
    return RanNum;
}
void FillMatrixWithRandomNumber(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            arr[i][j] = RandomNumber(1, 100);
        }
    }
}
void PrintMatrix(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            printf("%0*d ", 2, arr[i][j]);
        }
        cout << endl;
    }
}
int MaxNumberInMatrix(int arr[3][3], int Rows, int Cols)
{
    int MaxNum = arr[0][0];
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Cols; j++)
        {
            if (arr[i][j] > MaxNum)
            {
                MaxNum = arr[i][j];
            }
        }
    }

    return MaxNum;
}
int MinimumNumberInMatrix(int arr[3][3], int Rows, int Cols)
{
    int MinNum = arr[0][0];
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Cols; j++)
        {
            if (arr[i][j] < MinNum)
            {
                MinNum = arr[i][j];
            }
        }
    }

    return MinNum;
}
int main()
{
    srand((unsigned)time(NULL));

    int Matrix1[3][3];

    FillMatrixWithRandomNumber(Matrix1, 3, 3);
    cout << "Matrix 1 : " << endl;
    PrintMatrix(Matrix1, 3, 3);

    cout << "\nThe max number in matrix is : "<<MaxNumberInMatrix(Matrix1,3,3);
    cout << "\nThe minimum number in matrix is : " << MinimumNumberInMatrix(Matrix1, 3, 3);

    return 0;
}
