// 09 - Print Middle Row and Middle Column of a Matrix
#include <iostream>
#include <cmath>
#include <cstdio>

using namespace std;

int RandomNumber(int From, int To)
{
    int RanNum = rand() % (To - From + 1) + From;
    return RanNum;
}
void FillMatrixWithRandomNumber(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            arr[i][j] = RandomNumber(1, 10);
        }
    }
}
void PrintMatrix(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            printf(" %0*d", 2, arr[i][j]);
        }
        cout << endl;
    }
}
void PrintMiddleRowsOfMatrix(int arr[3][3], int Rows, int Cols)
{
    short MiddleRow = Rows / 2;

    for (int j = 0; j < Cols; j++)
    {
        printf(" %0*d ",2,arr[MiddleRow][j]);
    }
}
void PrintMiddleColsOfMatrix(int arr[3][3], int Rows, int Cols)
{
    short MiddleCols = Cols / 2;

    for (int i = 0; i < Rows; i++)
    {
        printf(" %0*d ", 2, arr[i][MiddleCols]);
    }
}
int main()
{
    srand((unsigned)time(NULL));

    int arr[3][3];

    FillMatrixWithRandomNumber(arr, 3, 3);
    cout << "\nThe matrix 3*3 : " << endl;
    PrintMatrix(arr, 3, 3);

    cout << "\nThe Middle Row matrix : " << endl;
    PrintMiddleRowsOfMatrix(arr, 3, 3);

    cout << "\nThe Middle Cols matrix : " << endl;
    PrintMiddleColsOfMatrix(arr, 3, 3);

    return 0;
}
