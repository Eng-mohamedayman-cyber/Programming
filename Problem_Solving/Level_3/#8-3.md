// 08 - Multiply Two Matrices
#include <iostream>
#include <cmath>
#include <cstdio>

using namespace std;

int RandomNumber(int From, int To)
{
    int RanNum = rand() % (To - From + 1) + From;
    return RanNum;
}
void FillMatrixWithRandomNumber(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            arr[i][j] = RandomNumber(1, 10);
        }
    }
}
void PrintMatrix(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            printf(" %0*d", 2, arr[i][j]);
        }
        cout << endl;
    }
}
void MultTwoMatrix(int arr[3][3], int arr2[3][3], int arrdestination[3][3], int Rows, int Cols)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Cols; j++)
        {
            arrdestination[i][j] = arr[i][j] * arr2[i][j];
        }
    }
}
int main()
{
    srand((unsigned)time(NULL));

    int arr[3][3];
    int arr2[3][3];
    int MultResult[3][3];
    
    FillMatrixWithRandomNumber(arr, 3, 3);
    cout << "\nThe matrix 1 : " << endl;
    PrintMatrix(arr, 3, 3);

    FillMatrixWithRandomNumber(arr2, 3, 3);
    cout << "\nThe matrix 2 : " << endl;
    PrintMatrix(arr2, 3, 3);

    MultTwoMatrix(arr, arr2, MultResult, 3, 3);

    cout << "\nThe matrix result : " << endl;
    PrintMatrix(MultResult, 3, 3);

    return 0;
}
