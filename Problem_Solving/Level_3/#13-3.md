// 13 - Check If a Matrix Is an Identity Matrix
#include <iostream>
#include <cmath>
#include <cstdio>

using namespace std;

void PrintMatrix(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            printf("%0*d ", 2, arr[i][j]);
        }
        cout << endl;
    }
}
bool AreMatrixIdentity(int Matrix1[3][3], int Rows, int Cols)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Cols; j++)
        {
            if ((i == j) && (Matrix1[i][j] != 1))
            {
                return false;
            }
            else if ((i != j) && (Matrix1[i][j] != 0))
            {
                return false;
            }
        }
    }
    return true;
}

int main()
{
    int Matrix1[3][3] = { {1,0,0},{0,1,0},{0,0,1} };

    cout << "Matrix 1 : " << endl;
    PrintMatrix(Matrix1, 3, 3);

    
    if (AreMatrixIdentity(Matrix1,  3, 3))
    {
        cout << "\nYes : Matrix Is Identity " << endl;
    }
    else
    {
        cout << "\nNo : Matrix Is NOT Identity " << endl;
    }

    return 0;
}
