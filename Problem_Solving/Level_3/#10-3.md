// 10 - Sum of All Elements in a Matrix
#include <iostream>
#include <cmath>
#include <cstdio>

using namespace std;

int RandomNumber(int From, int To)
{
    int RanNum = rand() % (To - From + 1) + From;
    return RanNum;
}
void FillMatrixWithRandomNumber(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            arr[i][j] = RandomNumber(1, 10);
        }
    }
}
void PrintMatrix(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            printf(" %0*d", 2, arr[i][j]);
        }
        cout << endl;
    }
}
int  SumOfAllMatrix(int arr[3][3], int Rows, int Cols)
{
    int sum = 0;

    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Cols; j++)
        {
            sum = sum + arr[i][j];
        }
    }

    return sum;
}

int main()
{
    srand((unsigned)time(NULL));

    int arr[3][3];

    FillMatrixWithRandomNumber(arr, 3, 3);
    cout << "\nThe matrix 3*3 : " << endl;
    PrintMatrix(arr, 3, 3);

    cout << "\nThe sum of matrix 3*3 = " <<SumOfAllMatrix(arr,3,3)<< endl;

    return 0;
}
