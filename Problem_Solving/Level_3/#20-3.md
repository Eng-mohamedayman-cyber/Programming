// 20 - Check If a Matrix Is Palindrome
#include <iostream>
#include <cmath>
#include <cstdio>

using namespace std;

int RandomNumber(int From, int To)
{
    int RanNum = rand() % (To - From + 1) + From;
    return RanNum;
}
void FillMatrixWithRandomNumber(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            arr[i][j] = RandomNumber(1, 100);
        }
    }
}
void PrintMatrix(int arr[3][3], int Rows, int Clos)
{
    for (int i = 0; i < Rows; i++)
    {
        for (int j = 0; j < Clos; j++)
        {
            printf("%0*d ", 2, arr[i][j]);
        }
        cout << endl;
    }
}
bool IsPalindromeMatrix(int arr[3][3], int Rows, int Cols)
{
    for (int i = 0; i < Rows; i++)
    {
        for (short j = 0; j < Cols/2; j++)
        {
            if (arr[i][j] != arr[i][Cols - 1 - j])
            {
                return false;
            }
        }
    }

    return true;
}
int main()
{
    srand((unsigned)time(NULL));

    int Matrix1[3][3] = { {1,2,1},{5,5,5},{7,3,7} };

//    FillMatrixWithRandomNumber(Matrix1, 3, 3);
    cout << "Matrix 1 : " << endl;
    PrintMatrix(Matrix1, 3, 3);

    if (IsPalindromeMatrix(Matrix1, 3, 3))
    {
        cout << "yes: Is Palindrome Matrix \n";
    }
    else
    {
        cout << "No: Is Not Palindrome Matrix \n";

    }

    return 0;
}
