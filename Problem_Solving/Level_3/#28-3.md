//#28-3 - Invert the Case of All Characters in a String from Uppercase to Lowercase and Vice Versa
#include <iostream>
#include <cmath>
#include <string>

using namespace std;

string ReadString()
{
    string S1;
    cout << "Please Enter Your String? " << endl;
    getline(cin, S1);

    return S1;
}
char InvertCharCase(char C1)
{
    return isupper(C1) ? tolower(C1) : toupper(C1);
}
string InvertAllCase(string S1)
{
    for (short i = 0; i <= S1.length(); i++)
    {
        S1[i] = InvertCharCase(S1[i]);
    }
    return S1;
}

int main()
{
    string S1 = ReadString();
    cout << "String After conversion : " << endl;

    S1 = InvertAllCase(S1);
    cout << S1 << endl;

    return 0;
}
